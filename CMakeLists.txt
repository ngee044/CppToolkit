cmake_minimum_required(VERSION 3.18)

include(CMakeDependentOption)

set(PROGRAM_NAME "CppToolkit")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED TRUE)
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -D_DEBUG")

set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

project(${PROGRAM_NAME} VERSION 1.0.0.0)

add_definitions(-DCPPHTTPLIB_OPENSSL_SUPPORT)

option(USE_ENCRYPT_LIBS "Use encrypt library" ON)
option(BUILD_THREAD_LIB "Build thread library" ON)
option(BUILD_DATABASE_LIB "Build database library" ON)
option(BUILD_NETWORK_LIB "Build network library" ON)
option(BUILD_SAMPLES "Build samples" ON)
option(BUILD_RABBIT_MQ_LIB "Build rabbitmq library" ON)
option(BUILD_REDIS_LIB "Build redis library" ON)
option(BUILD_KAFKA_LIB "Build kafka library" ON)

if(WIN32)
    option(BUILD_GAME_NETWORK_SESSION "Build game network session library" ON)
else()
    option(BUILD_GAME_NETWORK_SESSION "Build game network session library" OFF)
endif()

add_subdirectory(Utilities)
add_subdirectory(ThreadPool)
add_subdirectory(Database)

if (BUILD_GAME_NETWORK_SESSION)
    add_subdirectory(GameNetwork)
    add_subdirectory(GameDatabase)
endif()
if (BUILD_NETWORK_LIB)
    add_subdirectory(Network)
endif()
if (BUILD_REDIS_LIB)
    add_subdirectory(Redis)
endif()
if (BUILD_RABBIT_MQ_LIB)
    add_subdirectory(RabbitMQ)
endif()
if (BUILD_SAMPLES)
    add_subdirectory(Samples)
endif()
if (BUILD_KAFKA_LIB)
    add_subdirectory(Kafka)
endif()
